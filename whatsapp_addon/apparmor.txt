#include <tunables/global>

profile whatsapp_addon flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # The trusted Home Assistant base-image bootstrap uses S6 Overlay and Bashio.
  file,
  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  signal (send) set=(kill,term,int,hup,cont),

  # Trusted S6 overlay and shell bootstrap.
  /init rix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /sbin/** ix,
  /usr/sbin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio, runtime state, and temporary files.
  /usr/lib/bashio/** ix,
  /tmp/{,**} rwk,
  /data/{,**} rwk,
  /run.sh rix,

  # Constrain the long-lived, network-facing application.
  /usr/bin/node cx -> node,

  profile node flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>
    #include <abstractions/ssl_certs>

    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    signal (receive) peer=whatsapp_addon,
    signal (receive) peer=*_whatsapp_addon,

    # Runtime code and container metadata are read-only. Native modules may be
    # memory-mapped, and only temporary and persistent session data is writable.
    /** r,
    /usr/bin/node mr,
    /node_modules/{,**} mr,
    /tmp/{,**} rwk,
    /data/{,**} rwk,
  }
}
